inspect_trim()
├─ solve_trim()
│  ├─ _validate_trim_solve_options()
│  ├─ _pack_trim_solver_variables()
│  ├─ compute_trim_residual_vector()
│  ├─ _trim_residual_weights()
│  ├─ for iteration in [0, options.max_iterations)
│  │  ├─ _residual_norm_inf()
│  │  ├─ if residual_norm_inf <= options.residual_tolerance
│  │  │  └─ _build_trim_solution(converged = true)
│  │  ├─ weighted_residual = weights.cwiseProduct(residual)
│  │  ├─ jac_raw = compute_trim_residual_jac()
│  │  ├─ jac = weights.asDiagonal() * jac_raw
│  │  ├─ hess = jac.transpose() * jac + damping * I
│  │  ├─ gradient = jac.transpose() * weighted_residual
│  │  ├─ ldlt(hess)
│  │  ├─ step = ldlt.solve(-gradient)
│  │  ├─ if ldlt.info() != Eigen::Success || !step.allFinite()
│  │  │  └─ break
│  │  ├─ if step.norm() <= options.step_tolerance
│  │  │  └─ _build_trim_solution(converged = residual_norm_inf <= options.residual_tolerance)
│  │  ├─ residual_norm_2 = weighted_residual.norm()
│  │  ├─ step_scale = 1.0
│  │  ├─ accepted = false
│  │  ├─ while step_scale >= options.min_step_scale
│  │  │  ├─ z_trial = z + step_scale * step
│  │  │  ├─ compute_trim_residual_vector()
│  │  │  ├─ residual_trial_norm_2 = weights.cwiseProduct(residual_trial).norm()
│  │  │  ├─ if residual_trial_norm_2 < residual_norm_2
│  │  │  │  ├─ z = z_trial
│  │  │  │  ├─ residual = residual_trial
│  │  │  │  ├─ damping = max(options.initial_damping, damping / options.damping_growth)
│  │  │  │  └─ accepted = true
│  │  │  └─ else
│  │  │     └─ step_scale *= options.backtrack_scale
│  │  └─ if !accepted
│  │     └─ damping *= options.damping_growth
│  └─ _build_trim_solution(converged = false)
